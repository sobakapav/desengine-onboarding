Для компонента ввода (<Input>) в файле `mock.ts` важно показать разные состояния текста (пустой, заполненный, длинный) и интерактивные статусы (ошибка, заблокирован).

Здесь нам известно, что от пользователя ожидается email. В противном случае надо было бы указать и тип инпута.

``
export const mockData = [
//Обычный пустой инпут с подсказкой
{
	label: "Почта",
	type: email
	placeholder: "Введите почту",
	value: "",
	},
// Пользователь ввел почту
	{
	name: 'Электронная почта',
	type: 'email',
	placeholder: 'name@example.com',
	value: 'edu@eduhund.com',
	},
// Пользователь ввел почту неправильно
	{
	name: 'Ваш email',
	type: 'email',
	placeholder: 'Введите, пожалуйста, ваш электронный адрес',
	value: '1234567890',
	error: 'Введите корректный адрес',
	},
// Поле заблокировано (почту нельзя редактировать)
	{
	name: 'email',
	type: 'email',
	placeholder: 'Электронный адрес',
	value: 'test_email@company.com',
	disabled: true,
	}
];
``
